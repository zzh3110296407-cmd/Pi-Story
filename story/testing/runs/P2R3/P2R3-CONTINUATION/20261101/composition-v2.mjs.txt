import {existsSync,mkdirSync,readFileSync,writeFileSync,copyFileSync} from 'node:fs';
import path from 'node:path';
import {gzipSync} from 'node:zlib';
import {readEvidence,sha256,publicEvidence} from '../../../scripts/eval/p2/artifacts.ts';
import {scoreEvidenceV3} from '../../../scripts/eval/p2/score-v3.ts';
import {rescoreRun} from '../../../scripts/eval/p2/rescore.ts';
import {scanSecrets} from '../../../scripts/eval/p2/secrets.mjs';
const root=path.resolve('story/testing/runs/P2R3'),oldId='20261031T154922Z-102c3ed-r1',freeze='102c3edb352914934da701a4f6fb717fc610f46a';
const continuationIds=process.argv.slice(2);if(continuationIds.length!==2||new Set(continuationIds).size!==2)throw new Error('Both distinct continuation run IDs required');
const folders=[oldId,...continuationIds].map(id=>path.join(root,'P2-CHAP',id));
const manifests=folders.map(f=>JSON.parse(readFileSync(path.join(f,'run-manifest.json'),'utf8'))),segments=folders.map(f=>readEvidence(f));
if(manifests.some(x=>x.repositoryCommit!==freeze||x.dirty!==false||x.stage!=='P2R3'||x.suiteId!=='P2-CHAP'||x.repetition!==1))throw new Error('Frozen source/stage mismatch');
if(manifests[0].status!=='failed'||manifests[0].caseFailures.some(x=>x.caseId!=='527'||x.code!=='CHAPTER_TARGET_NOT_REACHED')||manifests.some(x=>x.fatalProviderError)||manifests.slice(1).some(x=>x.continuationOf!==oldId)||manifests[1].status!=='measured_with_failures'||manifests[1].caseFailures.some(x=>x.caseId!=='837'||x.code!=='CHAPTER_TARGET_NOT_REACHED')||manifests[2].status!=='measured_pending_gates'||manifests[2].caseFailures.length)throw new Error('Unexpected physical run outcome');
const expected=['1587','1769','42','527','837'].flatMap(id=>[1,2,3,4].map(n=>`${id}:${n}`));
function audit(rows){
 const chapters=rows.filter(x=>x.type==='chapter_result'),keys=chapters.map(x=>`${x.caseId}:${x.chapter}`);
 if(keys.length!==20||new Set(keys).size!==20||expected.some(k=>!keys.includes(k)))throw new Error('Expected exactly five prompts by four unique chapters');
 const requests=rows.filter(x=>x.type==='request').map(x=>x.requestId);
 if(new Set(requests).size!==requests.length)throw new Error('Request evidence duplicated');
 const macroCommits=rows.filter(x=>x.type==='tool_result'&&x.message?.details?.ok===true&&x.message.details.commit).map(x=>({key:`${x.caseId}:${x.message.details.commit.chapter}`,hash:sha256(x.message.details.commit.text)}));
 if(macroCommits.length!==20||chapters.some(x=>!macroCommits.some(c=>c.key===`${x.caseId}:${x.chapter}`&&c.hash===sha256(x.commit.text))))throw new Error('Chapter summary differs from actual successful macro');
 return {uniqueChapterCount:keys.length,keys,requests:requests.length,macroCommitsMatch:true};
}
const all=segments.flat(),coverage=audit(all);let duplicateRejected=false;
try{audit([...all,all.find(x=>x.type==='chapter_result')]);}catch{duplicateRejected=true;}if(!duplicateRejected)throw new Error('Duplicate chapter counterexample was accepted');
const runId=`${oldId}-complete`,folder=path.join(root,'P2R3-COMPOSITION',runId);
if(existsSync(folder))throw new Error('Composition already exists; immutable evidence must not be overwritten');
mkdirSync(path.join(folder,'raw'),{recursive:true});
const rows=[{...segments[0].find(x=>x.type==='meta'),runId,composition:true,physicalRunIds:manifests.map(x=>x.runId)}];
for(let i=0;i<segments.length;i++)for(const raw of segments[i]){
 const row={...raw,physicalRunId:manifests[i].runId};
 if(row.type==='meta')row.type='physical_meta';
 if(typeof row.rawFile==='string'){
  const relative=`segments/${i}/${row.rawFile}`,target=path.join(folder,'raw',relative);mkdirSync(path.dirname(target),{recursive:true});
  copyFileSync(path.join(folders[i],'raw',row.rawFile),target);row.rawFile=relative;
 }
 rows.push(row);
}
const text=rows.map(x=>JSON.stringify(x)).join('\n')+'\n';publicEvidence(text);
const bytes=gzipSync(text);writeFileSync(path.join(folder,'raw/evidence.jsonl.gz'),bytes,{flag:'wx'});
const checksums={evidenceSha256:sha256(bytes)};writeFileSync(path.join(folder,'raw/checksums.json'),JSON.stringify(checksums,null,2)+'\n');
const costs=Object.fromEntries(['knownUsd','reservedUsd','unknownCalls','requests','totalUsd'].map(k=>[k,manifests.reduce((sum,x)=>sum+x.costs[k],0)]));
const manifest={...manifests[0],runId,status:'measured_pending_gates',selectedCaseId:null,caseFailures:[],rawChecksums:checksums,costs,composition:{schema:'story/p2r3-continuation-composition@1',physicalRunIds:manifests.map(x=>x.runId),sourceManifests:folders.map(f=>({folder:path.relative(root,f).replaceAll('\\','/'),sha256:sha256(readFileSync(path.join(f,'run-manifest.json')))})),sourceRawChecksums:folders.map(f=>JSON.parse(readFileSync(path.join(f,'raw/checksums.json'),'utf8'))),coverage,duplicateCounterexampleRejected:duplicateRejected,allOriginalRowsRetained:true,paidRequestsInThisView:0,physicalOutcomes:manifests.map(x=>({runId:x.runId,status:x.status,caseFailures:x.caseFailures})),hardMetricSource:'Physical runs only; exclude this view from all-real to avoid duplicate requests',reason:'Author paused and resumed repetition one; no completed chapter replayed, no sample or scoring change'}};
writeFileSync(path.join(folder,'run-manifest.json'),JSON.stringify(manifest,null,2)+'\n',{flag:'wx'});
const metrics=scoreEvidenceV3(readEvidence(folder,true),[]);writeFileSync(path.join(folder,'metrics.json'),JSON.stringify(metrics,null,2)+'\n',{flag:'wx'});
const verify=rescoreRun(folder);if(!verify.equal)throw new Error('Frozen rescore mismatch');
const scan=scanSecrets([folder],process.cwd());writeFileSync(path.join(folder,'secret-scan.json'),JSON.stringify(scan,null,2)+'\n');if(!scan.passed)throw new Error('Composition secret scan failed');
console.log(JSON.stringify({folder,runId,coverage,originalFailuresRetained:rows.filter(x=>['run_failure','measurement_stop'].includes(x.type)).length,costs,rescoreEqual:true,metrics:metrics.metrics.filter(x=>!x.manual).map(({id,n,value,successes,passed})=>({id,n,value,successes,passed}))}));
